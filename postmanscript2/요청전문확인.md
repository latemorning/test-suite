```json
  "phub_tr_no": "{{v_phub_tr_no}}",        -- 약관 조회 응답값에 있음
  "shop_cd": "{{v_shop_cd}}",              -- 이전 단계에서 셋팅된 고정값
```

```javascript


// 1. 고객부담수수료율
// 계산하기 편하게 0 으로 셋팅하면 됩니다.
const shop_cmsn_rate = 0
pm.variables.set("v_shop_cmsn_rate", shop_cmsn_rate)



// 2. 상품금액
// 아래 use_points 배열에 입력한 금액의 합 입니다.
const ttl_pay_amt = 7000
pm.variables.set("v_ttl_pay_amt", ttl_pay_amt)



/* 각 카드사별 사용포인트
- 사용금액에 따라 아래 배열에서 수정
- 전단계 포인트 조회에서 조회가 되지 않은 카드사는 제외됨
- point를 0으로 하면 해당 카드사 포인트 사용요청 제외됨
*/
const use_points = [
{"pnt_cd": "bccard", "point": 1000},           // 
{"pnt_cd": "kbpointree", "point": 1000},       // 
{"pnt_cd": "shinhan", "point": 1000},
{"pnt_cd": "hanamembers", "point": 1000},
{"pnt_cd": "samsungcard", "point": 1000},      // 
{"pnt_cd": "hyundaicard", "point": 0},         // 계산 복잡해지므로 제외
{"pnt_cd": "lottemembers", "point": 1000},
{"pnt_cd": "wooricard", "point": 0},           // 현재 조회 안됨
{"pnt_cd": "nhmembers", "point": 1000},        // 
//    {"pnt_cd": "naverpay", "point": 1000},   // 현재 가맹점에 해당사항 없음
]

// 이전단계 포인트 조회 결과 배열을 가져옵니다.
const prvdr_list = pm.variables.get("v_prvdr_list")

// 3. 각 카드사 사용 point 총합
// 이전단계 포인트 조회에서 조회된 카드사만 필터링
// 위 use_points에 입력했어도 포인트 조회에서 조회되지 않았다면 무시됨
let ttl_pnt = 0
prvdr_list.forEach(prvdr => {
const use_point_obj = use_points.filter(use_point => use_point.pnt_cd === prvdr.pnt_cd)
ttl_pnt = ttl_pnt + use_point_obj[0].point
})
pm.variables.set("v_ttl_pnt", ttl_pnt)


// 4. 각 카드사 사용 point_amt 총합
// 카드사별 전환율이 적용된 금액 (전환율은 포인트 조회 단계에서 구함)
let ttl_pnt_amt = 0
prvdr_list.forEach(prvdr => {
const use_point_obj = use_points.filter(use_point => use_point.pnt_cd === prvdr.pnt_cd)
ttl_pnt_amt = ttl_pnt_amt + use_point_obj[0].point * prvdr.pnt_exch_rate
})
pm.variables.set("v_ttl_pnt_amt", ttl_pnt_amt)


// 5. 결제할인액
// 4번 카드사별 전환율이 적용된 금액에 가맹점 할인률 적용
let ttl_cprt_amt = ttl_pnt_amt - (ttl_pnt_amt * (shop_cmsn_rate / 100))  
pm.variables.set("v_ttl_cprt_amt", ttl_cprt_amt)


// 6. 잔여금액
const ttl_rmnd_amt = ttl_pay_amt - ttl_cprt_amt
pm.variables.set("v_ttl_rmnd_amt", ttl_rmnd_amt)


// 7. 카드사별 포인트사용 요청 list
// 포인트 조회단계에서 응답과 use_points에 입력한 값을 pnc_cd로 매칭해서 아래와 같은 결과를 만듬
// [{"pnt_tr_no":"BC250930000020927761","pnt_cd":"bccard","name":"비씨카드","point":150000,"point_amt":150000}]
const req_prvdr_list = prvdr_list.filter(p => use_points.some(u => p.pnt_cd === u.pnt_cd && u.point > 0)).map((prvdr) => {

    const rtn_obj = {}

    rtn_obj.pnt_tr_no = prvdr.pnt_tr_no
    rtn_obj.pnt_cd = prvdr.pnt_cd
    rtn_obj.name = prvdr.pnt_nm

    const use_point_obj = use_points.filter(use_point => use_point.pnt_cd === prvdr.pnt_cd)[0]

    console.log(JSON.stringify(use_point_obj))

    rtn_obj.point = use_point_obj.point
    rtn_obj.point_amt = use_point_obj.point * prvdr.pnt_exch_rate

    return rtn_obj
})
pm.variables.set("v_req_prvdr_list", JSON.stringify(req_prvdr_list))


// 8. hash  
// 아래 사용한 변수값 string을 합쳐서 javascript 암호화 모듈을 사용하여 hash 생성함
const pg_tr_no = pm.variables.get("v_pg_tr_no")
const phub_tr_no = pm.variables.get("v_phub_tr_no")
const shop_cd = pm.variables.get("v_shop_cd")

const hash_source = pg_tr_no + ttl_pnt_amt + shop_cd + phub_tr_no
const hash2 = CryptoJS.SHA256(hash_source).toString()
pm.variables.set("v_hash2", hash2)
```
